214 bool negotiate(const char* export_name) {
215 active = false;
216 if (export_name == nullptr) export_name = "";
217
218 // server hello: "NBDMAGIC", "IHAVEOPT", handshake flags
219 uint8_t hello[18];
220 if (!readFully(hello, sizeof(hello))) return fail("No server handshake");
221 if (getBE64(hello) != NBD_INIT_MAGIC) return fail("Invalid init magic");
222 if (getBE64(hello + 8) != NBD_OPTS_MAGIC)
223 return fail("Server does not support newstyle options");
224 uint16_t server_flags = getBE16(hello + 16);
225 if (!(server_flags & NBD_FLAG_FIXED_NEWSTYLE))
226 return fail("Server does not support fixed newstyle");
227 no_zeroes = (server_flags & NBD_FLAG_NO_ZEROES) != 0;
228
229 // client flags
230 uint8_t cflags[4];
231 putBE32(cflags, NBD_FLAG_C_FIXED_NEWSTYLE |
232 (no_zeroes ? NBD_FLAG_C_NO_ZEROES : 0));
233 if (!writeFully(cflags, sizeof(cflags)))
234 return fail("Could not send client flags");
235
236 size_t name_len = strlen(export_name);
237 if (name_len > NBD_MAX_NAME_LEN) return fail("Export name too long");
238
239 // defaults for servers which don't report block sizes
240 min_block_size = 1;
241 preferred_block_size = 4096;
242 max_payload = 0;
243
244 int rc = negotiateGo(export_name, name_len);
245 if (rc < 0) return fail("Export negotiation failed");
246 if (rc == 0) {
247 // the server refused NBD_OPT_GO: the connection is still in the
248 // option phase, so use NBD_OPT_EXPORT_NAME
249 NBD_LOGI("NBD_OPT_GO refused, using NBD_OPT_EXPORT_NAME");
250 if (!negotiateExportName(export_name, name_len))
251 return fail("Export negotiation failed");
252 }
253
254 handle_counter = 0;
255 last_error = NBD_OK;
256 active = true;
257 NBD_LOGI("Connected to export '%s': %llu bytes, block size %lu..%lu",
258 export_name, (unsigned long long)export_size,
259 (unsigned long)min_block_size,
260 (unsigned long)preferred_block_size);
261 return true;
262 }

267 int negotiateGo(const char* export_name, size_t name_len) {
268 // option data: name length, name, number of info requests, info types
269 uint8_t opt[16];
270 putBE64(opt, NBD_OPTS_MAGIC);
271 putBE32(opt + 8, NBD_OPT_GO);
272 putBE32(opt + 12, (uint32_t)(4 + name_len + 2 + 2));
273 uint8_t name_len_field[4];
274 putBE32(name_len_field, (uint32_t)name_len);
275 uint8_t info_req[4];
276 putBE16(info_req, 1); // one info request
277 putBE16(info_req + 2, NBD_INFO_BLOCK_SIZE);
278 if (!writeFully(opt, sizeof(opt)) ||
279 !writeFully(name_len_field, sizeof(name_len_field)) ||
280 !writeFully((const uint8_t*)export_name, name_len) ||
281 !writeFully(info_req, sizeof(info_req)))
282 return -1;
283
284 // replies: NBD_REP_INFO messages, then NBD_REP_ACK (or an error)
285 bool got_export = false;
286 while (true) {
287 uint8_t buf[16];
288 uint32_t type = 0;
289 size_t payload_len = 0;
290 if (!readOptionReply(NBD_OPT_GO, type, buf, sizeof(buf), payload_len))
291 return -1;
292 if (type == NBD_REP_ACK) return got_export ? 1 : -1;
293 if (type == NBD_REP_INFO) {
294 if (payload_len < 2) return -1;
295 uint16_t info = getBE16(buf);
296 if (info == NBD_INFO_EXPORT && payload_len >= 12) {
297 export_size = getBE64(buf + 2);
298 transmission_flags = getBE16(buf + 10);
299 got_export = true;
300 } else if (info == NBD_INFO_BLOCK_SIZE && payload_len >= 14) {
301 min_block_size = getBE32(buf + 2);
302 preferred_block_size = getBE32(buf + 6);
303 max_payload = getBE32(buf + 10);
304 }
305 continue;
306 }
307 if (type & NBD_REP_FLAG_ERROR) return 0;
308 return -1;
309 }
310 }

314 bool negotiateExportName(const char* export_name, size_t name_len) {
315 uint8_t opt[16];
316 putBE64(opt, NBD_OPTS_MAGIC);
317 putBE32(opt + 8, NBD_OPT_EXPORT_NAME);
318 putBE32(opt + 12, (uint32_t)name_len);
319 if (!writeFully(opt, sizeof(opt)) ||
320 !writeFully((const uint8_t*)export_name, name_len))
321 return false;
322
323 // reply: size (8), transmission flags (2), optional 124 zero bytes
324 uint8_t reply[10 + 124];
325 size_t reply_len = no_zeroes ? 10 : sizeof(reply);
326 if (!readFully(reply, reply_len)) return false;
327 export_size = getBE64(reply);
328 transmission_flags = getBE16(reply + 8);
329 return true;
330 }

334 bool readOptionReply(uint32_t option, uint32_t& type, uint8_t* buf,
335 size_t buf_size, size_t& payload_len) {
336 uint8_t hdr[20];
337 if (!readFully(hdr, sizeof(hdr))) return false;
338 if (getBE64(hdr) != NBD_REP_MAGIC) return false;
339 if (getBE32(hdr + 8) != option) return false;
340 type = getBE32(hdr + 12);
341 payload_len = getBE32(hdr + 16);
342 size_t keep = payload_len < buf_size ? payload_len : buf_size;
343 if (keep > 0 && !readFully(buf, keep)) return false;
344 return skip(payload_len - keep);
345 }

360 uint64_t sendRequest(uint16_t type, uint16_t flags, uint64_t offset,
361 uint64_t len, const uint8_t* payload) {
362 if (len > 0xFFFFFFFFULL) return 0;
363 uint64_t handle = ++handle_counter;
364 uint8_t req[NBD_REQUEST_SIZE];
365 putBE32(req, NBD_REQUEST_MAGIC);
366 putBE16(req + 4, flags);
367 putBE16(req + 6, type);
368 putBE64(req + 8, handle);
369 putBE64(req + 16, offset);
370 putBE32(req + 24, (uint32_t)len);
371 if (!writeFully(req, sizeof(req))) {
372 NBD_LOGE("Could not send request");
373 active = false;
374 return 0;
375 }
376 // only write requests carry data
377 if (type == NBD_CMD_WRITE && len > 0 && payload != nullptr) {
378 if (!writeFully(payload, (size_t)len)) {
379 NBD_LOGE("Could not send write data");
380 active = false;
381 return 0;
382 }
383 }
384 return handle;
385 }

388 bool receiveReply(uint64_t handle, uint8_t* data, size_t len) {
389 uint8_t hdr[NBD_SIMPLE_REPLY_SIZE];
390 if (!readFully(hdr, sizeof(hdr))) {
391 NBD_LOGE("No reply from server");
392 active = false;
393 return false;
394 }
395 if (getBE32(hdr) != NBD_SIMPLE_REPLY_MAGIC) {
396 NBD_LOGE("Invalid reply magic");
397 active = false;
398 return false;
399 }
400 if (getBE64(hdr + 8) != handle) {
401 NBD_LOGE("Unexpected reply handle");
402 active = false;
403 return false;
404 }
405 last_error = getBE32(hdr + 4);
406 if (last_error != NBD_OK) {
407 NBD_LOGE("Server returned error %lu", (unsigned long)last_error);
408 return false;
409 }
410 // the server only sends data for successful reads
411 if (data != nullptr && len > 0) {
412 if (!readFully(data, len)) {
413 NBD_LOGE("Could not read data");
414 active = false;
415 return false;
416 }
417 }
418 return true;
419 }

465 bool readFully(uint8_t* data, size_t len) {
466 uint32_t start = millis();
467 while (len > 0) {
468 int avail = client.available();
469 if (avail > 0) {
470 size_t req = (size_t)avail < len ? (size_t)avail : len;
471 int n = client.read(data, req);
472 if (n > 0) {
473 data += n;
474 len -= n;
475 start = millis();
476 continue;
477 }
478 }
479 if (!client.connected()) return false;
480 if (millis() - start > timeout_ms) {
481 NBD_LOGE("Read timeout");
482 return false;
483 }
484 delay(1);
485 }
486 return true;
487 }
488
489 bool writeFully(const uint8_t* data, size_t len) {
490 uint32_t start = millis();
491 while (len > 0) {
492 size_t n = client.write(data, len);
493 if (n > 0) {
494 data += n;
495 len -= n;
496 start = millis();
497 continue;
498 }
499 if (!client.connected()) return false;
500 if (millis() - start > timeout_ms) {
501 NBD_LOGE("Write timeout");
502 return false;
503 }
504 delay(1);
505 }
506 return true;
