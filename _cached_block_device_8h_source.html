202 Slot* load(uint64_t page) {
203 Slot* victim = &slots[0];
204 for (size_t i = 0; i < slot_count; i++) {
205 if (!slots[i].valid) {
206 victim = &slots[i];
207 break;
208 }
209 if (slots[i].last_used < victim->last_used) victim = &slots[i];
210 }
211 uint64_t start = page * page_size;
212 uint64_t dev_size = backend.size();
213 if (start >= dev_size) return nullptr;
214 uint64_t n = dev_size - start;
215 if (n > page_size) n = page_size;
216 victim->valid = false;
217 if (!backend.read(start, victim->data, (size_t)n)) return nullptr;
218 victim->page = page;
