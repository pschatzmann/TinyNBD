55 bool handshake(Client& client) {
56 current = nullptr;
57 if (!allocateBuffer()) {
58 NBD_LOGE("Not enough memory for buffer of %u bytes",
59 (unsigned)buffer_size);
60 return false;
61 }
62 uint8_t hello[18];
63 putBE64(hello, NBD_INIT_MAGIC);
64 putBE64(hello + 8, NBD_OPTS_MAGIC);
65 putBE16(hello + 16, NBD_FLAG_FIXED_NEWSTYLE | NBD_FLAG_NO_ZEROES);
66 if (!writeFully(client, hello, sizeof(hello))) return false;
67
68 uint8_t tmp[16];
69 if (!readFully(client, tmp, 4)) return false;
70 uint32_t client_flags = getBE32(tmp);
71 if (client_flags & ~(NBD_FLAG_C_FIXED_NEWSTYLE | NBD_FLAG_C_NO_ZEROES)) {
72 NBD_LOGE("Unsupported client flags: 0x%lx", (unsigned long)client_flags);
73 return false;
74 }
75 no_zeroes = client_flags & NBD_FLAG_C_NO_ZEROES;
76
77 while (true) {
78 if (!readFully(client, tmp, 16)) return false;
79 if (getBE64(tmp) != NBD_OPTS_MAGIC) {
80 NBD_LOGE("Invalid option magic");
81 return false;
82 }
83 uint32_t option = getBE32(tmp + 8);
84 uint32_t len = getBE32(tmp + 12);
85 NBD_LOGD("Option %lu, len %lu", (unsigned long)option,
86 (unsigned long)len);
87
88 // the option data is stored in the buffer
89 if (len > buffer_size - 1) {
90 if (!drain(client, len)) return false;
91 if (!sendOptionReply(client, option, NBD_REP_ERR_TOO_BIG)) return false;
92 continue;
93 }
94 uint8_t* data = buffer;
95 if (!readFully(client, data, len)) return false;
96
97 switch (option) {
98 case NBD_OPT_EXPORT_NAME: {
99 data[len] = 0;
100 current = findExport((const char*)data);
101 if (current == nullptr) {
102 NBD_LOGE("Unknown export: '%s'", (const char*)data);
103 return false;
104 }
105 uint8_t reply[10 + 124] = {0};
106 putBE64(reply, current->device->size());
107 putBE16(reply + 8, transmissionFlags());
108 if (!writeFully(client, reply, no_zeroes ? 10 : sizeof(reply)))
109 return false;
110 logStart();
111 return true;
112 }
113
114 case NBD_OPT_ABORT:
115 sendOptionReply(client, option, NBD_REP_ACK);
116 return false;
117
118 case NBD_OPT_LIST:
119 if (len != 0) {
120 if (!sendOptionReply(client, option, NBD_REP_ERR_INVALID))
121 return false;
122 break;
123 }
124 if (!sendExportList(client)) return false;
125 break;
126
127 case NBD_OPT_INFO:
128 case NBD_OPT_GO: {
129 int rc = processInfo(client, option, data, len);
130 if (rc < 0) return false;
131 if (rc > 0 && option == NBD_OPT_GO) {
132 logStart();
133 return true;
134 }
135 break;
136 }
137
138 default:
139 if (!sendOptionReply(client, option, NBD_REP_ERR_UNSUP))
140 return false;
141 break;
142 }
143 }

150 bool processRequest(Client& client) {
151 if (current == nullptr) return false;
152 uint8_t req[NBD_REQUEST_SIZE];
153 if (!readFully(client, req, sizeof(req))) return false;
154 if (getBE32(req) != NBD_REQUEST_MAGIC) {
155 NBD_LOGE("Invalid request magic");
156 return false;
157 }
158 uint16_t flags = getBE16(req + 4);
159 uint16_t type = getBE16(req + 6);
160 const uint8_t* handle = req + 8;
161 uint64_t offset = getBE64(req + 16);
162 uint32_t len = getBE32(req + 24);
163 BlockDevice& dev = *current->device;
164 uint64_t dev_size = dev.size();
165 bool in_range = offset <= dev_size && len <= dev_size - offset;
166 bool read_only = isReadOnly();
167 NBD_LOGD("Cmd %u flags 0x%x offset %llu len %lu", type, flags,
168 (unsigned long long)offset, (unsigned long)len);
169
170 switch (type) {
171 case NBD_CMD_READ:
172 if (!in_range) return sendReply(client, handle, NBD_EINVAL);
173 return processRead(client, handle, offset, len);
174
175 case NBD_CMD_WRITE: {
176 uint32_t error = NBD_OK;
177 if (read_only)
178 error = NBD_EPERM;
179 else if (!in_range)
180 error = NBD_ENOSPC;
181 if (error != NBD_OK) {
182 if (!drain(client, len)) return false;
183 return sendReply(client, handle, error);
184 }
185 if (!receiveWrite(client, dev, offset, len, error)) return false;
186 if (error == NBD_OK && (flags & NBD_CMD_FLAG_FUA) && !dev.flush())
187 error = NBD_EIO;
188 return sendReply(client, handle, error);
189 }
190
191 case NBD_CMD_DISC:
192 NBD_LOGI("Client disconnected");
193 dev.flush();
194 return false;
195
196 case NBD_CMD_FLUSH:
197 return sendReply(client, handle, dev.flush() ? NBD_OK : NBD_EIO);
198
199 case NBD_CMD_TRIM: {
200 uint32_t error = NBD_OK;
201 if (read_only)
202 error = NBD_EPERM;
203 else if (!dev.canTrim() || !in_range)
204 error = NBD_EINVAL;
205 else if (!dev.trim(offset, len))
206 error = NBD_EIO;
207 else if ((flags & NBD_CMD_FLAG_FUA) && !dev.flush())
208 error = NBD_EIO;
209 return sendReply(client, handle, error);
210 }
211
212 case NBD_CMD_WRITE_ZEROES: {
213 uint32_t error = NBD_OK;
214 if (read_only)
215 error = NBD_EPERM;
216 else if (!in_range)
217 error = NBD_ENOSPC;
218 else if (!dev.writeZeroes(offset, len))
219 error = NBD_EIO;
220 else if ((flags & NBD_CMD_FLAG_FUA) && !dev.flush())
221 error = NBD_EIO;
222 return sendReply(client, handle, error);
223 }
224
225 default:
226 NBD_LOGE("Unsupported command: %u", type);
227 return sendReply(client, handle, NBD_EINVAL);
228 }
229 }

298 int processInfo(Client& client, uint32_t option, uint8_t* data,
299 uint32_t len) {
300 if (len < 6) return replyResult(client, option, NBD_REP_ERR_INVALID);
301 uint32_t name_len = getBE32(data);
302 if (name_len > len - 6)
303 return replyResult(client, option, NBD_REP_ERR_INVALID);
304 uint16_t info_count = getBE16(data + 4 + name_len);
305 if (4 + name_len + 2 + 2 * (uint32_t)info_count != len)
306 return replyResult(client, option, NBD_REP_ERR_INVALID);
307
308 // copy the requested info types before we terminate the name
309 const uint8_t* infos = data + 4 + name_len + 2;
310 bool want_name = false, want_desc = false, want_block = false;
311 for (int j = 0; j < info_count; j++) {
312 uint16_t info = getBE16(infos + 2 * j);
313 if (info == NBD_INFO_NAME) want_name = true;
314 if (info == NBD_INFO_DESCRIPTION) want_desc = true;
315 if (info == NBD_INFO_BLOCK_SIZE) want_block = true;
316 }
317 char* name = (char*)data + 4;
318 name[name_len] = 0;
319
320 NBDExport* exp = findExport(name);
321 if (exp == nullptr) {
322 NBD_LOGE("Unknown export: '%s'", name);
323 return replyResult(client, option, NBD_REP_ERR_UNKNOWN);
324 }
325 current = exp;
326
327 uint8_t info[12];
328 putBE16(info, NBD_INFO_EXPORT);
329 putBE64(info + 2, exp->device->size());
330 putBE16(info + 10, transmissionFlags());
331 if (!sendOptionReply(client, option, NBD_REP_INFO, info, 12)) return -1;
332
333 if (want_name && !sendInfoString(client, option, NBD_INFO_NAME, exp->name))
334 return -1;
335 if (want_desc && exp->description != nullptr &&
336 !sendInfoString(client, option, NBD_INFO_DESCRIPTION,
337 exp->description))
338 return -1;
339 if (want_block) {
340 uint32_t min_bs = exp->device->minBlockSize();
341 uint32_t pref_bs = exp->device->preferredBlockSize();
342 if (pref_bs < 512) pref_bs = 512;
343 if (pref_bs < min_bs) pref_bs = min_bs;
344 uint8_t bs[14];
345 putBE16(bs, NBD_INFO_BLOCK_SIZE);
346 putBE32(bs + 2, min_bs);
347 putBE32(bs + 6, pref_bs);
348 putBE32(bs + 10, 32UL * 1024 * 1024); // max payload of 32 MiB
349 if (!sendOptionReply(client, option, NBD_REP_INFO, bs, sizeof(bs)))
350 return -1;
351 }
352 if (!sendOptionReply(client, option, NBD_REP_ACK)) return -1;
353 if (option == NBD_OPT_INFO) current = nullptr;
354 return 1;
355 }

421 bool processRead(Client& client, const uint8_t* handle, uint64_t offset,
422 uint32_t len) {
423 BlockDevice& dev = *current->device;
424 // read the first chunk before sending the header so that we can still
425 // report an error
426 size_t n = chunkSize(offset, len);
427 if (!dev.read(offset, buffer, n)) return sendReply(client, handle, NBD_EIO);
428 // send the header together with the first chunk
429 uint8_t* hdr = buffer - HEADER_ROOM;
430 setupReply(hdr, handle, NBD_OK);
431 if (!writeFully(client, hdr, HEADER_ROOM + n)) return false;
432 offset += n;
433 len -= n;
434 while (len > 0) {
435 n = chunkSize(offset, len);
436 if (!dev.read(offset, buffer, n)) {
437 // header was already sent: we can only disconnect
438 NBD_LOGE("Read failed at %llu", (unsigned long long)offset);
439 return false;
440 }
441 if (!writeFully(client, buffer, n)) return false;
442 offset += n;
443 len -= n;
444 }
445 return true;
446 }
447
449 bool receiveWrite(Client& client, BlockDevice& dev, uint64_t offset,
450 uint32_t len, uint32_t& error) {
451 while (len > 0) {
452 size_t n = chunkSize(offset, len);
453 if (!readFully(client, buffer, n)) return false;
454 if (error == NBD_OK && !dev.write(offset, buffer, n)) error = NBD_EIO;
455 offset += n;
456 len -= n;
457 }
458 return true;
459 }
460
462 bool drain(Client& client, uint32_t len) {
463 while (len > 0) {
464 size_t n = len < buffer_size ? len : buffer_size;
465 if (!readFully(client, buffer, n)) return false;
466 len -= n;
467 }
468 return true;
469 }

471 bool readFully(Client& client, uint8_t* data, size_t len) {
472 uint32_t start = millis();
473 while (len > 0) {
474 int avail = client.available();
475 if (avail > 0) {
476 size_t req = (size_t)avail < len ? (size_t)avail : len;
477 int n = client.read(data, req);
478 if (n > 0) {
479 data += n;
480 len -= n;
481 start = millis();
482 continue;
483 }
484 }
485 if (!client.connected()) return false;
486 if (millis() - start > timeout_ms) {
487 NBD_LOGE("Read timeout");
488 return false;
489 }
490 delay(1);
491 }
492 return true;
493 }
494
495 bool writeFully(Client& client, const uint8_t* data, size_t len) {
496 uint32_t start = millis();
497 while (len > 0) {
498 size_t n = client.write(data, len);
499 if (n > 0) {
500 data += n;
501 len -= n;
502 start = millis();
503 continue;
504 }
505 if (!client.connected()) return false;
506 if (millis() - start > timeout_ms) {
507 NBD_LOGE("Write timeout");
508 return false;
509 }
510 delay(1);
511 }
512 return true;
513 }
